cmake_minimum_required(VERSION 3.24)
project(maya_imma_q4_trial LANGUAGES CXX CUDA)
find_package(CUDAToolkit REQUIRED)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
if(NOT CMAKE_CUDA_ARCHITECTURES)
  set(CMAKE_CUDA_ARCHITECTURES 120)
endif()
get_filename_component(MAYA_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(MAYA_KERNELS_LIBRARY "${MAYA_ROOT}/build/strata_kernels.lib" CACHE FILEPATH "Existing unmodified kernels archive")
set(MAYA_CORE_LIBRARY "${MAYA_ROOT}/build/strata_core.lib" CACHE FILEPATH "Existing core archive")
add_executable(imma_q4_feasibility prototype.cu feasibility.cu)
target_include_directories(imma_q4_feasibility PRIVATE "${MAYA_ROOT}/include" "${MAYA_ROOT}/third_party/ggml")
target_link_libraries(imma_q4_feasibility PRIVATE "${MAYA_KERNELS_LIBRARY}" "${MAYA_CORE_LIBRARY}" CUDA::cudart_static)
set_target_properties(imma_q4_feasibility PROPERTIES CUDA_RUNTIME_LIBRARY Static)
target_compile_options(imma_q4_feasibility PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--ptxas-options=-v>)
